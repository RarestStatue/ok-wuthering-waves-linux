#     pip-compile --pip-args='--index-url https://pypi.org/simple --timeout 60 --retries 10' \

# that is the range spec, and pip-compile carries the branch straight through -- but a lock
# whose top dependency is a moving ref does not lock anything. Replace the @linux-port
# suffix on the ok-script line below with the fork commit the lock was resolved against:
#     git -C ../ok-script-linux rev-parse linux-port
